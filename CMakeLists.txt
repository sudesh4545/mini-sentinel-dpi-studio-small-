cmake_minimum_required(VERSION 3.20)
project(SentinelDPI VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(sentinel_core src/pcap.cpp src/parser.cpp src/rules.cpp src/engine.cpp)
target_include_directories(sentinel_core PUBLIC include)
if(MSVC)
  target_compile_options(sentinel_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(sentinel_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(sentinel-dpi src/main.cpp)
target_link_libraries(sentinel-dpi PRIVATE sentinel_core)
enable_testing()
add_executable(sentinel-tests tests/test_core.cpp)
target_link_libraries(sentinel-tests PRIVATE sentinel_core)
add_test(NAME sentinel-core-tests COMMAND sentinel-tests)
